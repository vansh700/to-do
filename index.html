Step 1 of 5: Create Google Sheet

1️⃣ Open your Google Sheets home page, and create a **new blank spreadsheet**.

Rename the sheet to something meaningful to you, like Gourav Life OS Sheet.

function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    
    // 1. Save Sync metadata
    var metaSheet = getOrCreateSheet(ss, "Sync_Metadata");
    metaSheet.clearContents();
    metaSheet.appendRow(["Key", "Value"]);
    metaSheet.appendRow(["Last Synced At", new Date().toISOString()]);
    metaSheet.appendRow(["Total Expense Records", data.expenses ? data.expenses.length : 0]);
    metaSheet.appendRow(["Total Gym Records", data.gym_logs ? data.gym_logs.length : 0]);
    metaSheet.appendRow(["Total Routine Logs", data.mr_daily_logs ? countKeys(data.mr_daily_logs) : 0]);
    
    // 2. Save Expenses
    if (data.expenses) {
      var sheet = getOrCreateSheet(ss, "Expenses");
      sheet.clearContents();
      sheet.appendRow(["ID", "Date", "Desc", "Category", "Amount", "Account", "Month"]);
      data.expenses.forEach(function(item) {
        sheet.appendRow([item.id, item.date, item.desc, item.cat, item.amt, item.acc, item.m]);
      });
    }
    
    // 3. Save Gym Logs
    if (data.gym_logs) {
      var sheet = getOrCreateSheet(ss, "Gym_Logs");
      sheet.clearContents();
      sheet.appendRow(["Date", "Workout Type", "Duration (min)", "Weight (kg)", "Calories (cal)", "Notes"]);
      data.gym_logs.forEach(function(item) {
        sheet.appendRow([
          item.date,
          item.type || "",
          item.dur || 0,
          item.wt || "",
          item.cal || "",
          item.notes || ""
        ]);
      });
    }
    
    // 4. Save Morning Routine Logs
    if (data.mr_daily_logs) {
      var sheet = getOrCreateSheet(ss, "Routine_Daily_Logs");
      sheet.clearContents();
      sheet.appendRow(["Date", "Activity", "Target", "Done", "Completed %"]);
      for (var date in data.mr_daily_logs) {
        var dayLog = data.mr_daily_logs[date];
        for (var act in dayLog) {
          var actLog = dayLog[act];
          sheet.appendRow([
            date,
            act,
            actLog.target || 0,
            actLog.done || 0,
            actLog.target ? Math.round((actLog.done / actLog.target) * 100) : 0
          ]);
        }
      }
    }
    
    // 5. Save Routine Targets
    if (data.mr_targets) {
      var sheet = getOrCreateSheet(ss, "Routine_Targets");
      sheet.clearContents();
      sheet.appendRow(["Activity", "Target (min)"]);
      var actNames = ["Meditation", "Yoga", "Pranayama", "Exercise / Gym", "Geeta Reading", "Study / Learning", "Journaling"];
      data.mr_targets.forEach(function(target, idx) {
        var name = actNames[idx] || ("Activity " + idx);
        sheet.appendRow([name, target]);
      });
    }
    
    return ContentService.createTextOutput(JSON.stringify({
      status: "success",
      message: "Sync completed successfully at " + new Date().toLocaleString()
    })).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({
      status: "error",
      message: err.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}

function doGet(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var result = { expenses: {}, gym_logs: [], mr_targets: [], mr_daily_logs: {} };
    
    function formatDate(d) {
      if (d instanceof Date) {
        var yyyy = d.getFullYear();
        var mm = String(d.getMonth() + 1).padStart(2, '0');
        var dd = String(d.getDate()).padStart(2, '0');
        return yyyy + '-' + mm + '-' + dd;
      }
      if (typeof d === 'string') {
        if (d.includes('T')) return d.split('T')[0];
        return d;
      }
      return d;
    }
    
    // Read Expenses sheet
    var expSheet = ss.getSheetByName("Expenses");
    if (expSheet && expSheet.getLastRow() > 1) {
      var expData = expSheet.getRange(2, 1, expSheet.getLastRow() - 1, 7).getValues();
      expData.forEach(function(row) {
        var mk = row[6]; // Month key
        var dateStr = formatDate(row[1]);
        if (!result.expenses[mk]) result.expenses[mk] = [];
        result.expenses[mk].push({
          id: row[0],
          date: dateStr,
          notes: row[2],
          cat: row[3],
          amount: Number(row[4]) || 0
        });
      });
    }
    
    // Read Gym Logs
    var gymSheet = ss.getSheetByName("Gym_Logs");
    if (gymSheet && gymSheet.getLastRow() > 1) {
      var gymData = gymSheet.getRange(2, 1, gymSheet.getLastRow() - 1, 6).getValues();
      gymData.forEach(function(row) {
        var dateStr = formatDate(row[0]);
        var type = row[1];
        var dur = Number(row[2]) || 0;
        var wt = row[3] !== "" ? Number(row[3]) : "";
        var cal = row[4] !== "" ? Number(row[4]) : "";
        var notes = row[5] || "";
        result.gym_logs.push({ date: dateStr, type: type, dur: dur, wt: wt, cal: cal, notes: notes });
      });
    }
    
    // Read Morning Routine Logs
    var routineSheet = ss.getSheetByName("Routine_Daily_Logs");
    if (routineSheet && routineSheet.getLastRow() > 1) {
      var routineData = routineSheet.getRange(2, 1, routineSheet.getLastRow() - 1, 4).getValues();
      routineData.forEach(function(row) {
        var dateStr = formatDate(row[0]);
        var act = row[1];
        var target = Number(row[2]) || 0;
        var done = Number(row[3]) || 0;
        if (dateStr && act) {
          if (!result.mr_daily_logs[dateStr]) result.mr_daily_logs[dateStr] = {};
          result.mr_daily_logs[dateStr][act] = { target: target, done: done };
        }
      });
    }
    
    // Read Routine Targets
    var targetSheet = ss.getSheetByName("Routine_Targets");
    var actNames = ["Meditation", "Yoga", "Pranayama", "Exercise / Gym", "Geeta Reading", "Study / Learning", "Journaling"];
    if (targetSheet && targetSheet.getLastRow() > 1) {
      var targetData = targetSheet.getRange(2, 1, targetSheet.getLastRow() - 1, 2).getValues();
      var targetsArray = [];
      targetData.forEach(function(row) {
        var name = row[0];
        var val = Number(row[1]);
        var idx = actNames.indexOf(name);
        if (idx !== -1) {
          targetsArray[idx] = val;
        }
      });
      for (var i = 0; i < actNames.length; i++) {
        if (targetsArray[i] === undefined) {
          var defaults = [20, 30, 15, 45, 15, 60, 10];
          targetsArray[i] = defaults[i];
        }
      }
      result.mr_targets = targetsArray;
    } else {
      result.mr_targets = [20, 30, 15, 45, 15, 60, 10];
    }
    
    // JSONP callback support
    var callback = e.parameter.callback;
    var output = JSON.stringify(result);
    if (callback) {
      return ContentService.createTextOutput(callback + '(' + output + ')').setMimeType(ContentService.MimeType.JAVASCRIPT);
    }
    return ContentService.createTextOutput(output).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({
      status: "error",
      message: err.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}

function getOrCreateSheet(ss, name) {
  var sheet = ss.getSheetByName(name);
  if (!sheet) {
    sheet = ss.insertSheet(name);
  }
  return sheet;
}

function countKeys(obj) {
  var count = 0;
  for (var k in obj) {
    if (obj.hasOwnProperty(k)) count++;
  }
  return count;
}
